const path = require('path');
const settings = require('../settings');
const loadConfig = require('../../config/load');
const { MT5VPSAdapter } = require('./client');
const { CommandJournal } = require('./commandJournal');
const events = require('../events');
const {followers,transferCopying}=require('./sizing');
const {normalizePrimaryRisk,primaryRisk}=require('./primaryRisk');
const {managementFreshness}=require('./managementFreshness');
settings.register('mt5-vps', path.join(__dirname,'config/mt5-vps.json'), path.join(__dirname,'config/mt5-vps-settings-descriptor.json'));
settings.onApply('mt5-vps', ({ changedPaths }) => ({restartRequiredPaths:changedPaths}));
function readConfig() { return loadConfig('../services/mt5-vps/config/mt5-vps.json'); }
function initService(api) {
  let adapter;
  const accountAdapters = new Map();
  let closing = false, watching = false;
  const {app, ipcMain, powerSaveBlocker} = require('electron');
  const runtimeActivity=require('./runtimeActivity').createRuntimeActivity(powerSaveBlocker);
  const tunnelRecovery=require('./tunnelRecovery').createTunnelRecovery(app.getPath('userData'));
  api.mt5VPS = {
    getAdapter() {
      // macOS safeStorage can cache an unusable key for this process when
      // decryption happens before Electron is ready. LT starts with services.
      if (app.isReady?.() === false) throw Error('ISCC запускается; подключение MT5 ожидает готовности приложения');
      if (!adapter) {
        const connection = new MT5VPSAdapter(readConfig());
        connection.commandJournal = new CommandJournal(path.join(app.getPath('userData'), 'mt5-command-journal.json'), {asyncWrites:true});
        connection.commandJournal.prepareEntryPlans?.();
        connection.events = events;
        connection.quoteLogFile = path.join(app.getPath('userData'),'logs','mt5-quote-stream.jsonl');
        connection.on('quote:updated', snapshot => {
          const symbol = snapshot.symbol === connection.config.spxSymbol ? 'SPX500' : 'USTEC';
          api.instrumentInfo?.ingest?.({provider:'mt5-vps',symbol},snapshot);
        });
        connection.on('quote:stale',symbols => {
          const selected=symbols?.map(symbol=>symbol===connection.config.spxSymbol ? 'SPX500' : 'USTEC') || ['SPX500','USTEC'];
          for (const symbol of selected) api.instrumentInfo?.ingest?.({provider:'mt5-vps',symbol},{});
        });
        adapter = connection;
        runtimeActivity.start(connection.config.enabled);
      }
      api.execution?.wireAdapter?.(adapter, 'mt5-vps');
      return adapter;
    }
  };
  function forAccount(account) {
    const primary = api.mt5VPS.getAdapter();
    if (!account || String(account)===String(primary.config.account)) return primary;
    if (!followers(primary.config).some(row=>row.account===String(account)) && !primary.remoteAccounts?.includes(String(account))) throw Error('Счёт не настроен');
    if (!accountAdapters.has(String(account))) {
      const connection = new MT5VPSAdapter({...primary.config, account:String(account),routePrefix:`/accounts/${account}`,mirrorAccount:null,followers:[]});
      connection.quoteLogFile=primary.quoteLogFile;
      connection.commandJournal = new CommandJournal(path.join(app.getPath('userData'),`mt5-command-journal-${account}.json`), {asyncWrites:true});
      connection.connectionRestarting=primary.connectionRestarting===true;
      accountAdapters.set(String(account),connection);
    }
    return accountAdapters.get(String(account));
  }
  function acceptManagement(state) {
    const primary = api.mt5VPS.getAdapter();
    primary.remoteAccounts = Object.keys(state.accountStates).filter(value=>value!==String(primary.config.account));
    for (const [account,health] of Object.entries(state.accountStates)) {
      if(primary.managementCache?.isCurrent(account,health)===false)continue;
      const connection = forAccount(account);
      if (healthyManagementAccount(health,account,primary.config.server))
        connection.acceptManagementState(health,connection===primary ? state : health);
      else if(health.connected===false && health.connectionState!=='checking')connection.managementSnapshot=null;
    }
  }
  function healthyManagementAccount(health,account,server) {
    return health?.connected===true && health.evidenceReady!==false && String(health.account)===String(account)
      && health.server===server && Array.isArray(health.orders) && Array.isArray(health.capabilities);
  }
  function managementDisplayAccounts(state,primary) {
    const display=managementFreshness(state,Date.now());
    return Object.fromEntries(Object.entries(display.accountStates).map(([account,health])=>{
      const native=forAccount(account).managementNativeState;
      const older=primary.managementCache?.isCurrent(account,health)===false
        || native?.managementEpoch===health.managementEpoch && Number.isFinite(native?.checkedAt)
          && Number.isFinite(health.checkedAt) && health.checkedAt<native.checkedAt;
      return [account,older ? {...health,connected:false,evidenceReady:false,reason:'Проверяем актуальность позиций счёта'} : health];
    }));
  }
  let receiptRefresh, receiptRefreshAt=0;
  const receiptAttempts=new WeakMap();
  function refreshManagementReceipt(primary,connection,accountStates) {
    const now=Date.now();
    if(receiptRefresh || now-receiptRefreshAt<10000)return;
    const views=Object.values(accountStates);
    const flat=views.length>0 && views.every(view=>view.evidenceReady!==false && view.connected===true
      && Array.isArray(view.orders) && view.orders.length===0);
    const parents=new Set(views.flatMap(view=>view.orders || []).filter(row=>row.entryPlan?.id)
      .map(row=>'entry:'+row.entryPlan.id));
    const candidates=[];
    for(const current of new Set([connection,primary])) {
      const rows=accountStates[String(current.config.account)]?.orders || [];
      const tickets=new Set(rows.map(row=>String(row.ticket)));
      for(const record of current.commandJournal.records) {
        if(record.result.status!=='unknown' || current.inFlight?.has(record.ticket))continue;
        const active=tickets.has(record.ticket) || current===primary && parents.has(record.ticket);
        const recent=now-record.at>=0 && now-record.at<=300000;
        const last=receiptAttempts.get(record) || 0;
        if((active || recent || flat) && now-last>=60000)candidates.push({current,record,active:active || recent,last});
      }
    }
    candidates.sort((a,b)=>Number(b.active)-Number(a.active) || a.last-b.last || b.record.at-a.record.at);
    const next=candidates[0];
    if(!next)return;
    const {current,record}=next;
    receiptRefreshAt=now;
    receiptAttempts.set(record,now);
    // Keep receipt recovery off the position read, and do not change the
    // coordinator's position-view signature to check an unrelated command.
    receiptRefresh=Promise.resolve().then(()=>current.commandJournal.reconcile(async id=>
      current.validateCommandReceipt(id,(await current.request(`/receipt?id=${encodeURIComponent(id)}`)).result),{id:record.id}))
      .catch(()=>{}).finally(()=>{receiptRefresh=null;});
  }
  const warmthTimer = require('timers').setInterval(async () => {
    if (closing || watching || !adapter?.config.enabled) return;
    watching = true;
    try {
      // Discover positions even with the dialog closed. This GET never arms or
      // resends a command and never subscribes followers to price streams.
      {
        const state = await adapter.maintainManagementSnapshot();
        if (closing) return;
        if (state) acceptManagement(state);
      }
      await Promise.all([adapter,...accountAdapters.values()].map(connection=>connection.maintainExecutionWarmth()));
    } catch { /* Existing UI status and VPS checks handle a disconnected terminal. */ }
    finally { watching = false; }
  },1000);
  warmthTimer.unref?.();
  // before-quit can be cancelled while the TV connector is stopping. Keep the
  // authenticated quote stream alive until all windows have actually closed.
  app.once('will-quit',() => { closing=true; runtimeActivity.stop(); require('timers').clearInterval(warmthTimer); adapter?.close(); for (const connection of accountAdapters.values()) connection.close(); });
  require('../brokerage/brokerageAdapters')['mt5-vps'] = () => api.mt5VPS.getAdapter();
  const enrollment=require('./accountEnrollment').createAccountEnrollment({
    getAdapter:()=>api.mt5VPS.getAdapter(),isTradingBusy:()=>Boolean(api.execution?.isTradingBusy?.())
      || [...accountAdapters.values()].some(connection=>connection.inFlight.size || connection.connectionRestarting)
  });
  ipcMain.handle('mt5-vps:account-catalog',()=>enrollment.run('catalog'));
  ipcMain.handle('mt5-vps:account-discovery',()=>enrollment.run('discover'));
  ipcMain.handle('mt5-vps:account-register',(_evt,payload)=>enrollment.run('register',payload));
  ipcMain?.handle?.('mt5-vps:status', async () => {
    const config = readConfig();
    if (!config.enabled) return { connected:false,message:'MT5 VPS: выключено' };
    let connection;
    try {
      connection = api.mt5VPS.getAdapter();
      const state=await connection.managementState({maxAgeMs:2000,allowPartial:true});
      acceptManagement(state);
      const primary=state.accountStates?.[String(config.account)];
      if(!primary)throw Error('Нет состояния основного счёта');
      const copies=followers(config).filter(row=>row.enabled);
      const accounts=[{account:String(config.account)},...copies].map(({account})=>{
        const health=state.accountStates?.[account];
        const status=healthyManagementAccount(health,account,config.server) ? 'ready'
          : health?.connectionState==='unavailable' ? 'unavailable' : 'checking';
        return {account,status};
      });
      connection.lastStatus={connected:true,message:`MT5 VPS · ${config.account}`,at:Date.now(),
        accounts,build:primary.build,clientBuild:require('./buildInfo').readBuildInfo()};
      if(connection.capabilities.includes('quote-stream'))connection.ensureQuoteStream();
      const degraded=accounts.filter(row=>row.status!=='ready');
      if(degraded.length)connection.lastStatus.message+=' · '+degraded.map(row=>`${row.account}: ${row.status==='checking' ? 'проверка' : 'недоступен'}`).join(', ');
      else if(copies.length)connection.lastStatus.message+=' · счета '+copies.map(row=>`${row.account} ×${row.multiplier}`).join(', ');
      const deadline=require('./timeExit').timeExitConfig(config);
      if (deadline.enabled) connection.lastStatus.message += ` · закрытие ${deadline.time} NY`;
      const risk = api.riskManager?.snapshot?.().config;
      if (accounts[0].status==='ready' && risk?.enabled && risk.providers?.['mt5-vps']?.enabled) await connection.listOpenOrders().catch(()=>{});
    } catch (error) { return {connected:false,message:error.message,canRecoverConnection:await tunnelRecovery.available(config)}; }
    return {...connection.lastStatus,canRecoverConnection:await tunnelRecovery.available(config)};
  });
  let reconnecting;
  ipcMain.handle('mt5-vps:reconnect',async () => {
    if (reconnecting) return reconnecting;
    if (enrollment.busy) return {status:'rejected',reason:'Дождись завершения добавления счёта'};
    let primary;
    try {primary=api.mt5VPS.getAdapter();} catch(error) {return {status:'rejected',reason:error.message};}
    if (api.execution?.isTradingBusy?.() || [primary,...accountAdapters.values()].some(connection=>connection.inFlight.size)) return {status:'rejected',reason:'Дождись завершения отправленной торговой команды'};
    for (const connection of [primary,...accountAdapters.values()]) connection.connectionRestarting=true;
    reconnecting=(async()=>{
      try {
        await tunnelRecovery.reconnect(primary.config);
        const deadline=Date.now()+15000;let lastError;
        do {
          try {
            await primary.health();
            await Promise.all(followers(primary.config).filter(row=>row.enabled).map(row=>forAccount(row.account).health()));
            return {status:'ok',message:'Связь восстановлена · все включённые счета готовы'};
          } catch(error) {lastError=error;await new Promise(resolve=>setTimeout(resolve,500));}
        } while(Date.now()<deadline);
        throw lastError;
      } catch(error) {return {status:'rejected',reason:error?.message || 'Нет связи с VPS. Проверь интернет и доступ к Связке ключей'};}
      finally {for(const connection of [primary,...accountAdapters.values()]) connection.connectionRestarting=false;reconnecting=null;}
    })();
    return reconnecting;
  });
  ipcMain.handle('mt5-vps:trade-settings',async()=>{
    try {
      const primary=api.mt5VPS.getAdapter(),config=readConfig();
      const state=await primary.managementState({maxAgeMs:2000});
      acceptManagement(state);
      const health=state.accountStates?.[String(config.account)];
      if(!health?.connected)throw Error(health?.reason || 'Нет связи с основным счётом');
      // Settings need capabilities/account ids, not ticket histories,
      // reconciliation, command journals or a management execution lease.
      return {status:'ok',account:config.account,accounts:[config.account,...primary.remoteAccounts],
        followers:followers(config),primaryRisk:primaryRisk(config),
        timeExit:require('./timeExit').timeExitConfig(config),capabilities:state.capabilities || health.capabilities || []};
    } catch(error){return {status:'rejected',reason:error.message};}
  });
  ipcMain.handle('mt5-vps:verify-candle-symbol',async(_event,symbol)=>{
    try {
      require('../tvListener/candle-mappings').brokerName(symbol);
      const adapter=api.mt5VPS.getAdapter();
      const [row]=await adapter.validateCandleSymbols([symbol]);
      return {status:'ok',symbol:row.name,description:row.description};
    } catch(error) {return {status:'rejected',reason:error.message};}
  });
  async function chartRecorder(payload) {
    try {
      const primary=api.mt5VPS.getAdapter();
      if(!primary.capabilities?.includes('chart-recorder-v1'))throw Error('Требуется обновление приёмника VPS');
      if(payload!==undefined && (!payload || typeof payload.account!=='string' || !/^[1-9][0-9]{0,19}$/.test(payload.account)
        || typeof payload.revision!=='string' || !/^[a-f0-9]{32}$/.test(payload.revision)))throw Error('Обнови список терминалов и выбери счёт');
      const result=await primary.request('/chart-recorder',payload===undefined ? undefined : {account:payload.account,revision:payload.revision});
      return {status:'ok',chartRecorder:result.chartRecorder};
    }catch(error){return {status:'rejected',reason:error.message};}
  }
  ipcMain.handle('mt5-vps:chart-recorder',()=>chartRecorder());
  ipcMain.handle('mt5-vps:save-chart-recorder',(_evt,payload)=>payload===undefined
    ? {status:'rejected',reason:'Выбери счёт MT5'} : chartRecorder(payload));
  async function tradeHubRoles(payload) {
    try {
      const primary=api.mt5VPS.getAdapter();
      if(!primary.capabilities?.includes('tradehub-roles-v1'))throw Error('Требуется обновление приёмника VPS');
      if(payload!==undefined && (!payload || !['history','ticks','chart'].includes(payload.role)
        || typeof payload.account!=='string' || !/^[1-9][0-9]{0,19}$/.test(payload.account)
        || typeof payload.revision!=='string' || !/^[a-f0-9]{32}$/.test(payload.revision)))throw Error('Обнови список ролей и выбери счёт');
      const result=await primary.request('/tradehub-roles',payload===undefined ? undefined
        : {role:payload.role,account:payload.account,revision:payload.revision});
      return {status:'ok',tradeHubRoles:result.tradeHubRoles};
    }catch(error){return {status:'rejected',reason:error.message};}
  }
  ipcMain.handle('mt5-vps:tradehub-roles',()=>tradeHubRoles());
  ipcMain.handle('mt5-vps:save-tradehub-role',(_evt,payload)=>payload===undefined
    ? {status:'rejected',reason:'Выбери роль и счёт MT5'} : tradeHubRoles(payload));
  ipcMain.handle('mt5-vps:orders', async (_evt, account) => {
    try {
      const connection = forAccount(account);
      const primary = api.mt5VPS.getAdapter();
      const selected=String(connection.config.account);
      let state;
      try {
        state=await primary.managementState({maxAgeMs:1000,allowPartial:true});
      } catch(error) {
        if(error.code!=='MT5_VIEW_PENDING')throw error;
      }
      // Allow one handoff for the selected account's pending read only.
      if(!state || state.accountStates?.[selected]?.evidenceReady===false) {
        await new Promise(resolve=>setTimeout(resolve,120));
        try {state=await primary.managementState({maxAgeMs:0,allowPartial:true});}
        catch(error){if(!state)throw error;}
      }
      const remoteAccounts=primary.remoteAccounts || [];
      acceptManagement(state);
      const accountStates=managementDisplayAccounts(state,primary),health=accountStates[selected];
      const config=readConfig(),copies=followers(config);
      const accounts=[...new Set([config.account,...copies.map(row=>row.account),...remoteAccounts,
        ...(primary.remoteAccounts || []),...Object.keys(accountStates)].map(String))];
      const alerts=state.alerts || [];
      if (!healthyManagementAccount(health,selected,primary.config.server))
        return {status:'rejected',account:selected,server:primary.config.server,
          reason:health?.reason || 'Нет свежих данных выбранного счёта',accounts,accountStates,alerts};
      connection.managementLeaseUntil = Date.now()+10000;
      connection.maintainExecutionWarmth().catch(()=>{});
      refreshManagementReceipt(primary,connection,accountStates);
      const parentCommands=primary.commandJournal.records.filter(command=>['entry-protection','entry-close','entry-partial-close','entry-take'].includes(command.operation));
      const rules=(health.rules || []).map(rule=>({...rule,executionResults:Object.fromEntries(Object.entries(accountStates).flatMap(([account,accountState])=>{
        const id=account===String(health.account) ? rule.id : rule.id+':'+account;
        const matched=(accountState.rules || []).find(candidate=>candidate.id===id);
        return matched?.result?.timings ? [[account,matched.result]] : [];
      }))}));
      return {status:'ok',account:health.account,accounts,followers:copies,timeExit:require('./timeExit').timeExitConfig(config),timeRules:health.timeRules || [],capabilities:state.capabilities || health.capabilities || [],orders:health.orders,rules,
        alerts,accountStates,parentTakes:state.parentTakes || [],checkedAt:health.checkedAt,
        commands:[...new Map([...connection.commandJournal.records,...parentCommands].map(command=>[command.id,command])).values()]
          .sort((a,b)=>(b.at || 0)-(a.at || 0)).slice(0,100)};
    } catch (error) { return {status:'rejected',reason:error.message}; }
  });
  ipcMain.handle('mt5-vps:minute-bars',async(_evt,symbol)=>{
    try{return {status:'ok',data:await api.mt5VPS.getAdapter().minuteBars(symbol)};}
    catch(error){return {status:'rejected',reason:error.message};}
  });
  ipcMain.handle('mt5-vps:prepare-entry',async () => {
    try {
      const connection=api.mt5VPS.getAdapter();
      if (!connection.config.enabled) throw Error('MT5 VPS выключено');
      connection.managementLeaseUntil=Date.now()+20000;
      // Local LT configuration is available even while a network read waits.
      // Warming is an optimization; its result never grants entry authority.
      Promise.resolve().then(()=>connection.warmExecutionConnection()).catch(()=>{
        connection.recordQuoteStream?.('preparation','background-warm-failed');
      });
      return {ok:true,groups:api.levelTrack?.snapshot?.().config.groups || []};
    } catch(error) {return {ok:false,reason:error.message};}
  });
  function persistFollowers(primary,config,next) {
    const changed={...config,followers:next,mirrorAccount:Number(next.find(row=>row.enabled)?.account || 0)};
    if(!settings.writeConfig('mt5-vps',changed))throw Error('Не удалось сохранить настройки');
    primary.config.followers=next;
    primary.config.mirrorAccount=changed.mirrorAccount;
    return {status:'ok',followers:next};
  }
  ipcMain.handle('mt5-vps:save-followers',async (_evt,rows)=>{
    try {
      if(enrollment.busy)throw Error('Дождись завершения добавления счёта');
      const primary=api.mt5VPS.getAdapter(), config=readConfig();
      const next=followers({...config,followers:rows});
      const health=await primary.health();
      if (!health.capabilities?.includes('scaled-accounts')) throw Error('Обновите приёмник VPS');
      if(next.some(row=>row.enabled && row.riskPercent!=null) && !health.capabilities?.includes('entry-balance-risk-v1'))throw Error('Обновите VPS для риска от баланса');
      if (next.some(row=>!health.linkedAccounts?.includes(row.account))) throw Error('Сначала подключите отдельный терминал и зарегистрируйте счёт на VPS');
      const current=readConfig();
      if(enrollment.busy || String(current.account)!==String(config.account)
        || JSON.stringify(followers(current))!==JSON.stringify(followers(config))) {
        throw Error('Настройки счетов изменились. Обнови список');
      }
      return persistFollowers(primary,current,next);
    } catch (error) {return {status:'rejected',reason:error.message};}
  });
  ipcMain.handle('mt5-vps:transfer-copying',async (_evt,request)=>{
    try {
      if(enrollment.busy)throw Error('Дождись завершения добавления счёта');
      const primary=api.mt5VPS.getAdapter();
      const state=await primary.managementState({maxAgeMs:0});
      const current=readConfig(),capabilities=state.capabilities || [];
      if(enrollment.busy || String(primary.config.account)!==String(current.account))throw Error('Настройки счетов изменились. Обнови список');
      if(!capabilities.includes('scaled-accounts'))throw Error('Обнови приёмник VPS');
      const next=transferCopying(current,request,Object.keys(state.accountStates));
      const target=state.accountStates[request.to];
      if(!target?.connected || target.evidenceReady===false || target.server!==current.server || String(target.account)!==request.to) {
        throw Error('Нет подтверждённого подключения нового счёта');
      }
      if(request.riskPercent!=null && !capabilities.includes('entry-balance-risk-v1'))throw Error('Обнови VPS для риска от баланса');
      return persistFollowers(primary,current,next);
    } catch (error) {return {status:'rejected',reason:error.message};}
  });
  ipcMain.handle('mt5-vps:save-primary-risk',(_evt,value)=>{
    try {
      const primary=api.mt5VPS.getAdapter(),config=readConfig();
      if (String(value?.account)!==String(config.account) || String(primary.config.account)!==String(config.account))
        throw Error('Основной счёт изменился. Обнови настройки MT5');
      const risk=normalizePrimaryRisk(value);
      const policies={...config.primaryRiskByAccount,[String(config.account)]:risk};
      if (!settings.writeConfig('mt5-vps',{...config,primaryRiskByAccount:policies}))
        throw Error('Не удалось сохранить риск основного счёта');
      primary.config.primaryRiskByAccount=policies;
      return {status:'ok',primaryRisk:risk};
    } catch(error) {return {status:'rejected',reason:error.message};}
  });
  ipcMain.handle('mt5-vps:save-time-exit',async (_evt,value)=>{
    try {
      const primary=api.mt5VPS.getAdapter(),config=readConfig();
      const timeExit=require('./timeExit').timeExitConfig({timeExit:value});
      if (timeExit.enabled) {
        const state=await primary.managementState({maxAgeMs:0});
        for (const account of [String(config.account),...followers(config).filter(x=>x.enabled).map(x=>x.account)]) {
          const health=state.accountStates?.[account];
          if (!health?.connected || !health.capabilities?.includes('time-exit')) throw Error('Обновите и подключите приёмник на каждом счёте');
        }
        require('./timeExit').deadlineForNewTrade({timeExit});
      }
      const current=readConfig();
      if (timeExit.enabled && (String(current.account)!==String(config.account)
        || JSON.stringify(followers(current))!==JSON.stringify(followers(config)))) {
        throw Error('Настройки счетов изменились. Проверь время закрытия ещё раз');
      }
      if (!settings.writeConfig('mt5-vps',{...current,timeExit})) throw Error('Не удалось сохранить время закрытия');
      primary.config.timeExit=timeExit;
      return {status:'ok',timeExit};
    } catch(error) {return {status:'rejected',reason:error.message};}
  });
  ipcMain.handle('mt5-vps:manage', async (_evt, payload = {}) => {
    if(enrollment.busy)return {status:'rejected',reason:'Дождись завершения добавления счёта'};
    const connection = forAccount(payload.account);
    if(payload.action==='entry-take' && payload.target?.entryPlan)
      return api.mt5VPS.getAdapter().configureParentTake(payload.target,payload.take);
    if (['entry-close','entry-partial-close'].includes(payload.action) && payload.target?.entryPlan) {
      const result=await api.mt5VPS.getAdapter().closeEntryPosition({id:payload.target.entryPlan.id,symbol:payload.target.symbol}, {
        clickAt:payload.target.clickAt,...(payload.action==='entry-partial-close'?{closePercent:Number(payload.closePercent)}:{})
      });
      try {if(result.entry)_evt?.sender?.send('mt5-vps:entry-control',result);} catch { /* A closed view cannot change the durable command result. */ }
      return result;
    }
    if (payload.action === 'entry-protection' || payload.action === 'modify' && payload.target?.entryPlan) {
      const result=await api.mt5VPS.getAdapter().modifyEntryProtection(payload.target, payload.changes || {}, {
        account:String(connection.config.account),operation:payload.operation || 'modify'
      });
      try {if(result.entry)_evt?.sender?.send('mt5-vps:entry-control',result);} catch { /* A closed view cannot change the durable command result. */ }
      return result;
    }
    if (payload.action === 'cancel' && payload.target?.__isPosition === false) {
      return connection.cancelPendingOrder(payload.target.ticket, payload.target.symbol,payload.target.singleAccount,payload.target.clickAt);
    }
    if (payload.action === 'close' && payload.target?.__isPosition === true) {
      return connection.closePosition(payload.target, 'manual MT5 management', payload.volume,payload.percent);
    }
    if (payload.action === 'modify' && payload.target) return connection.modifyOrder(payload.target, payload.changes || {});
    if (payload.action === 'schedule-close' && payload.target?.__isPosition) return connection.schedulePartialClose(payload.target,payload.volume,payload.targetPrice,payload.comparison,payload.percent);
    if (payload.action === 'cancel-close' && payload.target) return connection.cancelScheduledClose(payload.target,payload.ruleId);
    return {status:'rejected',reason:'Неизвестная команда управления MT5'};
  });
}
